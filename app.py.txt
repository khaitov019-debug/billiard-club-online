from flask import Flask, render_template, request, jsonify
import sqlite3
import os
from datetime import datetime

app = Flask(__name__)
DB = "billiard_club.db"

DEFAULT_PRODUCTS = [
    ("Suv 0.5L", 3000, "💧"),
    ("Suv 1L", 5000, "💧"),
    ("Suv 1.5L", 7000, "💧"),
    ("Suv 2L", 9000, "💧"),
    ("Cola 0.5L", 7000, "🥤"),
    ("Cola 1L", 10000, "🥤"),
    ("Cola 1.5L", 13000, "🥤"),
    ("Cola 2L", 16000, "🥤"),
    ("Chips", 10000, "🍟"),
    ("Pista", 12000, "🥜"),
]

def get_db():
    db = sqlite3.connect(DB)
    db.row_factory = sqlite3.Row
    return db

def init_db():
    db = get_db()

    db.execute("""
        CREATE TABLE IF NOT EXISTS settings (
            key TEXT PRIMARY KEY,
            value TEXT
        )
    """)

    db.execute("""
        CREATE TABLE IF NOT EXISTS products (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            price INTEGER NOT NULL,
            icon TEXT DEFAULT '🍽️'
        )
    """)

    db.execute("""
        CREATE TABLE IF NOT EXISTS sales (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            category TEXT,
            item TEXT,
            amount INTEGER,
            created_at TEXT
        )
    """)

    settings = {
        "billiard_price": "30000",
        "tennis_price": "30000",
        "vip_price": "50000",
        "stick_price": "5000",
        "background": "default"
    }

    for key, value in settings.items():
        db.execute(
            "INSERT OR IGNORE INTO settings(key,value) VALUES(?,?)",
            (key, value)
        )

    count = db.execute(
        "SELECT COUNT(*) FROM products"
    ).fetchone()[0]

    if count == 0:
        db.executemany(
            "INSERT INTO products(name,price,icon) VALUES(?,?,?)",
            DEFAULT_PRODUCTS
        )

    db.commit()
    db.close()

init_db()


@app.route("/")
def index():
    return render_template("index.html")


@app.get("/api/state")
def state():
    db = get_db()

    settings = {
        row["key"]: row["value"]
        for row in db.execute("SELECT * FROM settings")
    }

    products = [
        dict(row)
        for row in db.execute(
            "SELECT * FROM products ORDER BY id"
        )
    ]

    sales = [
        dict(row)
        for row in db.execute(
            "SELECT * FROM sales ORDER BY id DESC LIMIT 500"
        )
    ]

    db.close()

    return jsonify({
        "settings": settings,
        "products": products,
        "sales": sales
    })


@app.post("/api/settings")
def save_settings():
    data = request.json or {}
    db = get_db()

    for key, value in data.items():
        db.execute("""
            INSERT INTO settings(key,value)
            VALUES(?,?)
            ON CONFLICT(key)
            DO UPDATE SET value=excluded.value
        """, (key, str(value)))

    db.commit()
    db.close()

    return jsonify({"ok": True})


@app.post("/api/products")
def add_product():
    data = request.json or {}

    name = data.get("name", "Yangi mahsulot")
    price = int(data.get("price", 0))
    icon = data.get("icon", "🍽️")

    db = get_db()

    cursor = db.execute(
        "INSERT INTO products(name,price,icon) VALUES(?,?,?)",
        (name, price, icon)
    )

    db.commit()
    product_id = cursor.lastrowid
    db.close()

    return jsonify({
        "ok": True,
        "id": product_id
    })


@app.put("/api/products/<int:product_id>")
def edit_product(product_id):
    data = request.json or {}

    db = get_db()

    db.execute("""
        UPDATE products
        SET name=?, price=?, icon=?
        WHERE id=?
    """, (
        data.get("name", ""),
        int(data.get("price", 0)),
        data.get("icon", "🍽️"),
        product_id
    ))

    db.commit()
    db.close()

    return jsonify({"ok": True})


@app.delete("/api/products/<int:product_id>")
def delete_product(product_id):
    db = get_db()

    db.execute(
        "DELETE FROM products WHERE id=?",
        (product_id,)
    )

    db.commit()
    db.close()

    return jsonify({"ok": True})


@app.post("/api/sale")
def add_sale():
    data = request.json or {}

    category = data.get("category", "other")
    item = data.get("item", "")
    amount = int(data.get("amount", 0))

    db = get_db()

    db.execute("""
        INSERT INTO sales(category,item,amount,created_at)
        VALUES(?,?,?,?)
    """, (
        category,
        item,
        amount,
        datetime.now().isoformat(timespec="seconds")
    ))

    db.commit()
    db.close()

    return jsonify({"ok": True})
