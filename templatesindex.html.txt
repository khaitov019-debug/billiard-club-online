<script>

let DATA = {
    settings:{},
    products:[],
    sales:[]
};

let runningTables = {};

function money(value){

    return Number(value || 0)
        .toLocaleString("uz-UZ") + " so'm";

}


async function loadData(){

    const response = await fetch("/api/state");

    DATA = await response.json();

    render();

}


function showPage(id, button){

    document
        .querySelectorAll(".page")
        .forEach(page => page.classList.remove("active"));

    document
        .getElementById(id)
        .classList.add("active");

    document
        .querySelectorAll(".nav-btn")
        .forEach(btn => btn.classList.remove("active"));

    button.classList.add("active");

}


function render(){

    document.getElementById("billiardPrice").value =
        DATA.settings.billiard_price || 30000;

    document.getElementById("tennisPrice").value =
        DATA.settings.tennis_price || 30000;

    document.getElementById("vipPrice").value =
        DATA.settings.vip_price || 50000;

    document.getElementById("stickPrice").value =
        DATA.settings.stick_price || 5000;

    document.getElementById("backgroundInput").value =
        DATA.settings.background || "default";


    renderTables(
        "billiardTables",
        5,
        "Billiard",
        Number(DATA.settings.billiard_price || 30000)
    );


    renderTables(
        "tennisTables",
        3,
        "Tennis",
        Number(DATA.settings.tennis_price || 30000)
    );


    renderProducts();

    renderReports();

}


function renderTables(containerId,count,type,price){

    const container =
        document.getElementById(containerId);

    container.innerHTML = "";


    for(let i=1;i<=count;i++){

        const key = type + i;

        const active = runningTables[key];

        const card = document.createElement("div");

        card.className = "card table-card";


        let timeText = "00:00:00";

        if(active){

            timeText =
                formatTime(
                    Date.now() - active.start
                );

        }


        card.innerHTML = `

            <div class="small">
                ${type.toUpperCase()}
            </div>

            <h3>№ ${i}</h3>

            <span class="status ${active ? "busy" : ""}">
                ${active ? "ISHLEYAPTI" : "BO‘SH"}
            </span>

            <div
                class="big"
                id="timer-${key}">
                ${timeText}
            </div>

            <div class="money">
                ${money(price)} / soat
            </div>

            <div class="buttons">

                ${
                    active
                    ?
                    `
                    <button
                        class="btn red"
                        onclick="stopTable('${type}',${i},${price})">
                        ⏹ To‘xtatish
                    </button>
                    `
                    :
                    `
                    <button
                        class="btn green"
                        onclick="startTable('${type}',${i})">
                        ▶ Boshlash
                    </button>
                    `
                }

            </div>

        `;

        container.appendChild(card);

    }

}


function formatTime(milliseconds){

    let totalSeconds =
        Math.floor(milliseconds / 1000);

    let hours =
        Math.floor(totalSeconds / 3600);

    totalSeconds %= 3600;

    let minutes =
        Math.floor(totalSeconds / 60);

    let seconds =
        totalSeconds % 60;


    return String(hours).padStart(2,"0")
        + ":"
        + String(minutes).padStart(2,"0")
        + ":"
        + String(seconds).padStart(2,"0");

}


function startTable(type,number){

    runningTables[type + number] = {
        start:Date.now()
    };

    render();

}


async function stopTable(type,number,price){

    const key = type + number;

    const table = runningTables[key];

    if(!table){
        return;
    }


    const hours =
        (Date.now() - table.start) / 3600000;


    const amount =
        Math.max(
            Math.ceil(hours * price),
            Math.ceil(price / 60)
        );


    await fetch("/api/sale",{

        method:"POST",

        headers:{
            "Content-Type":"application/json"
        },

        body:JSON.stringify({

            category:type,

            item:type + " №" + number,

            amount:amount

        })

    });


    delete runningTables[key];

    await loadData();

}


function renderProducts(){

    const container =
        document.getElementById("productList");

    container.innerHTML = "";


    DATA.products.forEach(product => {

        const card =
            document.createElement("div");

        card.className =
            "card product-card";


        card.innerHTML = `

            <div>

                <div class="product-icon">
                    ${product.icon}
                </div>

                <h3>
                    ${escapeHtml(product.name)}
                </h3>

                <div class="money">
                    ${money(product.price)}
                </div>

            </div>

            <button
                class="btn blue"
                onclick="sellProduct(${product.id})">

                🛒 Sotish

            </button>

        `;


        container.appendChild(card);

    });


    const manage =
        document.getElementById("manageProducts");

    manage.innerHTML = "";


    DATA.products.forEach(product => {

        const row =
            document.createElement("div");

        row.className =
            "manage-row";


        row.innerHTML = `

            <input
                id="name-${product.id}"
                value="${escapeAttr(product.name)}">

            <input
                id="price-${product.id}"
                type="number"
                value="${product.price}">

            <input
                id="icon-${product.id}"
                value="${escapeAttr(product.icon)}">

            <button
                class="btn green"
                onclick="editProduct(${product.id})">

                Saqlash

            </button>

            <button
                class="btn red"
                onclick="deleteProduct(${product.id})">

                O‘chirish

            </button>

        `;


        manage.appendChild(row);

    });

}


async function sellProduct(id){

    const product =
        DATA.products.find(
            p => Number(p.id) === Number(id)
        );

    if(!product){
        return;
    }


    await fetch("/api/sale",{

        method:"POST",

        headers:{
            "Content-Type":"application/json"
        },

        body:JSON.stringify({

            category:"Mahsulot",

            item:product.name,

            amount:product.price

        })

    });


    await loadData();

}


async function addProduct(){

    const name =
        document.getElementById("newName").value.trim();

    const price =
        Number(document.getElementById("newPrice").value);

    const icon =
        document.getElementById("newIcon").value.trim()
        || "🍽️";


    if(!name || !price){

        alert("Mahsulot nomi va narxini kiriting.");

        return;

    }


    await fetch("/api/products",{

        method:"POST",

        headers:{
            "Content-Type":"application/json"
        },

        body:JSON.stringify({
            name:name,
            price:price,
            icon:icon
        })

    });


    document.getElementById("newName").value = "";
    document.getElementById("newPrice").value = "";
    document.getElementById("newIcon").value = "";


    await loadData();

}


async function editProduct(id){

    const name =
        document.getElementById("name-" + id).value;

    const price =
        document.getElementById("price-" + id).value;

    const icon =
        document.getElementById("icon-" + id).value;


    await fetch("/api/products/" + id,{

        method:"PUT",

        headers:{
            "Content-Type":"application/json"
        },

        body:JSON.stringify({
            name:name,
            price:price,
            icon:icon
        })

    });


    await loadData();

}


async function deleteProduct(id){

    if(!confirm("Bu mahsulot o‘chirilsinmi?")){
        return;
    }


    await fetch("/api/products/" + id,{
        method:"DELETE"
    });


    await loadData();

}


async function saveSettings(){

    const data = {

        billiard_price:
            document.getElementById("billiardPrice").value,

        tennis_price:
            document.getElementById("tennisPrice").value,

        vip_price:
            document.getElementById("vipPrice").value,

        stick_price:
            document.getElementById("stickPrice").value

    };


    await fetch("/api/settings",{

        method:"POST",

        headers:{
            "Content-Type":"application/json"
        },

        body:JSON.stringify(data)

    });


    await loadData();

    alert("Narxlar saqlandi.");

}


async function saveBackground(){

    const background =
        document.getElementById("backgroundInput").value;


    await fetch("/api/settings",{

        method:"POST",

        headers:{
            "Content-Type":"application/json"
        },

        body:JSON.stringify({
            background:background
        })

    });


    await loadData();

    alert("Background sozlamasi saqlandi.");

}


function renderReports(){

    const now = new Date();

    const today =
        new Date(
            now.getFullYear(),
            now.getMonth(),
            now.getDate()
        );

    const week =
        new Date(today);

    week.setDate(
        today.getDate() - 6
    );


    const month =
        new Date(
            now.getFullYear(),
            now.getMonth(),
            1
        );


    let dayTotal = 0;
    let weekTotal = 0;
    let monthTotal = 0;


    DATA.sales.forEach(sale => {

        const date =
            new Date(sale.created_at);


        const amount =
            Number(sale.amount || 0);


        if(date >= today){
            dayTotal += amount;
        }

        if(date >= week){
            weekTotal += amount;
        }

        if(date >= month){
            monthTotal += amount;
        }

    });


    document.getElementById(
        "todayMoney"
    ).textContent = money(dayTotal);


    document.getElementById(
        "dayTotal"
    ).textContent = money(dayTotal);


    document.getElementById(
        "weekTotal"
    ).textContent = money(weekTotal);


    document.getElementById(
        "monthTotal"
    ).textContent = money(monthTotal);


    const table =
        document.getElementById("salesTable");


    table.innerHTML = "";


    DATA.sales
        .slice(0,50)
        .forEach(sale => {

            const row =
                document.createElement("tr");


            row.innerHTML = `

                <td>
                    ${sale.created_at
                        .replace("T"," ")}
                </td>

                <td>
                    ${escapeHtml(sale.category)}
                </td>

                <td>
                    ${escapeHtml(sale.item)}
                </td>

                <td class="money">
                    ${money(sale.amount)}
                </td>

            `;


            table.appendChild(row);

        });

}


function escapeHtml(value){

    return String(value)
        .replaceAll("&","&amp;")
        .replaceAll("<","&lt;")
        .replaceAll(">","&gt;")
        .replaceAll('"',"&quot;")
        .replaceAll("'","&#039;");

}


function escapeAttr(value){

    return escapeHtml(value);

}


setInterval(() => {

    Object.keys(runningTables)
        .forEach(key => {

            const element =
                document.getElementById(
                    "timer-" + key
                );

            if(element){

                element.textContent =
                    formatTime(
                        Date.now() -
                        runningTables[key].start
                    );

            }

        });

},1000);


loadData();
